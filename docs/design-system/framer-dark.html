08

Tokens als CSS Variables

Direkt übernehmbar — die Seite selbst ist mit exakt diesen Tokens gebaut.

:root {
  /* Backgrounds */
  --bg-app: #050505;   --bg-panel: #0F0F0F;   --bg-surface: #161616;
  --bg-elevated: #1C1C1C;   --bg-input: #232323;   --bg-hover: #2A2A2A;   --bg-active: #333333;

  /* Text */
  --text-primary: #F5F5F5;   --text-secondary: #999999;
  --text-tertiary: #5C5C5C;   --text-disabled: #3D3D3D;

  /* Borders */
  --border-subtle: rgba(255,255,255,.07);   --border-default: rgba(255,255,255,.12);   --border-strong: rgba(255,255,255,.20);

  /* Accent & Semantic */
  --accent: #2B6FF2;   --accent-hover: #4A85F5;   --accent-muted: rgba(43,111,242,.16);
  --teal: #8ED1CE;   --teal-deep: #57979B;
  --success: #3ECF8E;   --warning: #F5A623;   --danger: #F24B4B;

  /* Radius */
  --r-xs: 6px;  --r-sm: 8px;  --r-md: 10px;  --r-lg: 14px;  --r-xl: 20px;  --r-pill: 999px;

  /* Elevation */
  --shadow-sm: 0 1px 2px rgba(0,0,0,.4);
  --shadow-md: 0 8px 24px rgba(0,0,0,.5);
  --shadow-lg: 0 16px 48px rgba(0,0,0,.6), 0 0 0 1px rgba(255,255,255,.06);

  /* Typography — Inter Variable via rsms.me/inter */
  --font-sans: "InterVariable", "Inter", -apple-system, sans-serif;
  --font-mono: "SF Mono", "JetBrains Mono", Menlo, monospace;
}
